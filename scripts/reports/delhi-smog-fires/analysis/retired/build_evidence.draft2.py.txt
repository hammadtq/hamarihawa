def mw_tab(name):
    out = {}
    for y in sorted({r['season'] for r in MW if r['control'] == name}) + ['all']:
        s = [r for r in MW if r['control'] == name and (y == 'all' or r['season'] == y)]
        a = st.mean(float(r['pm']) for r in s); c = st.mean(float(r['control_pm']) for r in s)
        out[y] = dict(days=len(s), fire_weeks_mean=round(a), matched_mean=round(c), difference=round(a - c), share_of_fire_weeks_pct=round(100 * (a - c) / a))
    return out

import random
def linked(s):
    few = [r['ex'] for r in s if grp(r) == 'few']; many = [r for r in s if grp(r) == 'many']
    if len(few) < 5 or len(many) < 5: return None
    b = st.mean(few); allx = st.mean(r['ex'] for r in s) - b; allp = st.mean(r['pm'] for r in s); mx = st.mean(r['ex'] for r in many) - b; mp = st.mean(r['pm'] for r in many)
    return allx, 100 * allx / allp, mx, 100 * mx / mp
